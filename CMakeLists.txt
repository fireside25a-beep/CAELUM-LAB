cmake_minimum_required(VERSION 3.18)
project(CaelumNativePortingSDK VERSION 0.4.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
include(GNUInstallDirs)
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
if(UNIX AND NOT APPLE)
 set(CMAKE_INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
elseif(APPLE)
 set(CMAKE_INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
endif()
file(GLOB CP_IDENTITY_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.c" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.h.in" "${CMAKE_CURRENT_SOURCE_DIR}/adapters/*.c" "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h")
list(APPEND CP_IDENTITY_FILES "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT CP_IDENTITY_FILES)
set(CP_SOURCE_MATERIAL "")
foreach(input IN LISTS CP_IDENTITY_FILES)
 file(SHA256 "${input}" digest)
 file(RELATIVE_PATH name "${CMAKE_CURRENT_SOURCE_DIR}" "${input}")
 string(APPEND CP_SOURCE_MATERIAL "${name}:${digest}\n")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CP_IDENTITY_FILES})
string(SHA256 CP_SOURCE_HASH "${CP_SOURCE_MATERIAL}")
string(TOUPPER "${CMAKE_BUILD_TYPE}" CP_BUILD_TYPE_UPPER)
file(SHA256 "${CMAKE_C_COMPILER}" CP_COMPILER_SHA256)
string(SHA256 CP_BUILD_HASH "${CP_SOURCE_HASH};${CMAKE_C_COMPILER_ID};${CMAKE_C_COMPILER_VERSION};${CP_COMPILER_SHA256};${CMAKE_SYSTEM_NAME};${CMAKE_SYSTEM_PROCESSOR};${CMAKE_SIZEOF_VOID_P};${CMAKE_C_COMPILER_TARGET};${CMAKE_C_FLAGS};${CMAKE_C_FLAGS_${CP_BUILD_TYPE_UPPER}};${CMAKE_EXE_LINKER_FLAGS};${CMAKE_SHARED_LINKER_FLAGS};${CMAKE_BUILD_TYPE}")
configure_file(src/identity.h.in generated/identity.h @ONLY)
option(CP_BUILD_TESTS "Build and register native verification tests" ON)
add_library(caelum_port SHARED src/core.c src/sha256.c src/cpu.c src/adapter_loader.c src/exchange.c src/onnx.c src/science.c src/lab.c src/dynamics.c src/io.c src/locale_numeric.c src/numeric.c src/random_v2.c src/identity.c src/expression.c src/causal.c src/json.c src/protocol.c src/engine.c src/uncertainty.c)
target_include_directories(caelum_port PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_definitions(caelum_port PRIVATE CP_BUILD_SHARED)
set_target_properties(caelum_port PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 1 C_VISIBILITY_PRESET hidden)
if(NOT MSVC)
  target_compile_options(caelum_port PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror -fno-fast-math -ffp-contract=off)
  target_link_libraries(caelum_port PRIVATE m ${CMAKE_DL_LIBS})
else()
  target_compile_options(caelum_port PRIVATE /W4)
endif()
add_library(cp_linear_exchange MODULE adapters/linear_exchange.c src/locale_numeric.c)
target_link_libraries(cp_linear_exchange PRIVATE caelum_port)
target_include_directories(cp_linear_exchange PRIVATE include src)
set_target_properties(cp_linear_exchange PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden)
if(NOT MSVC)
  target_compile_options(cp_linear_exchange PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
endif()
add_executable(caelum-port src/cli.c)
target_link_libraries(caelum-port PRIVATE caelum_port)
if(NOT MSVC)
 target_compile_options(caelum-port PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror -fno-fast-math -ffp-contract=off)
endif()
if(CP_BUILD_TESTS)
 enable_testing()
 add_executable(caelum-port-tests tests/native_tests.c)
 target_link_libraries(caelum-port-tests PRIVATE caelum_port)
 add_test(NAME native_contracts COMMAND caelum-port-tests $<TARGET_FILE:cp_linear_exchange>)
 add_test(NAME platform_probe COMMAND caelum-port probe)
 add_executable(caelum-exchange-tests tests/exchange_tests.c)
 target_link_libraries(caelum-exchange-tests PRIVATE caelum_port)
 add_test(NAME exchange_contracts COMMAND caelum-exchange-tests)
 add_executable(caelum-science-tests tests/science_tests.c)
 target_link_libraries(caelum-science-tests PRIVATE caelum_port)
 if(NOT MSVC)
   target_link_libraries(caelum-science-tests PRIVATE m)
 endif()
 add_test(NAME science_contracts COMMAND caelum-science-tests)
 add_executable(caelum-dynamics-tests tests/dynamics_tests.c)
 target_link_libraries(caelum-dynamics-tests PRIVATE caelum_port)
 if(NOT MSVC)
   target_link_libraries(caelum-dynamics-tests PRIVATE m)
 endif()
 add_test(NAME dynamics_contracts COMMAND caelum-dynamics-tests)
 add_executable(caelum-parser-fuzz verification/parser_fuzz.c src/json.c src/locale_numeric.c)
 target_include_directories(caelum-parser-fuzz PRIVATE include src)
 if(NOT MSVC)
  target_link_libraries(caelum-parser-fuzz PRIVATE m)
 endif()
 add_test(NAME parser_mutations COMMAND caelum-parser-fuzz)
 add_executable(caelum-lab-tests tests/lab_tests.c)
 target_link_libraries(caelum-lab-tests PRIVATE caelum_port)
 if(NOT MSVC)
  target_link_libraries(caelum-lab-tests PRIVATE m)
 endif()
 add_test(NAME lab_engine_contracts COMMAND caelum-lab-tests ${CMAKE_CURRENT_BINARY_DIR})
 add_executable(caelum-release-regressions tests/release_regressions.c)
 target_link_libraries(caelum-release-regressions PRIVATE caelum_port)
 if(NOT MSVC)
  target_link_libraries(caelum-release-regressions PRIVATE m)
 endif()
 add_test(NAME release_regressions COMMAND caelum-release-regressions)
 add_executable(caelum-safety-tests tests/safety_tests.c)
 target_link_libraries(caelum-safety-tests PRIVATE caelum_port)
 foreach(case basis_alias basis_atomic holm_overlap matrix_span dlpack_flags arrow_dictionary csv_nul midpoint_branch)
  add_test(NAME safety_${case} COMMAND caelum-safety-tests ${case})
 endforeach()
 add_executable(caelum-science-consumer examples/science_consumer.c)
 target_link_libraries(caelum-science-consumer PRIVATE caelum_port)
 if(NOT MSVC)
   target_link_libraries(caelum-science-consumer PRIVATE m)
 endif()
 add_test(NAME actual_science_consumer COMMAND caelum-science-consumer)
 add_executable(caelum-consumer-example examples/consumer.c)
 target_link_libraries(caelum-consumer-example PRIVATE caelum_port)
 add_test(NAME actual_sdk_consumer COMMAND caelum-consumer-example)
endif()

install(TARGETS caelum_port caelum-port cp_linear_exchange
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/caelum_port.h include/caelum_exchange.h include/caelum_science.h include/caelum_lab.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/caelum)
# Relocatable pkg-config metadata: do not freeze the CMake configure prefix.
file(RELATIVE_PATH CP_PC_PREFIX_REL "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_PREFIX}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/caelum-port.pc
"prefix=\${pcfiledir}/${CP_PC_PREFIX_REL}\nlibdir=\${pcfiledir}/..\nincludedir=\${prefix}/${CMAKE_INSTALL_INCLUDEDIR}/caelum\n\nName: caelum-port\nDescription: CAELUM native scientific model interchange\nVersion: ${PROJECT_VERSION}\nLibs: -L\${libdir} -lcaelum_port\nCflags: -I\${includedir}\n")
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/caelum-port.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)

option(CP_BUILD_WORKBENCH "Build the native X11 scientific workbench on Linux" OFF)
if(CP_BUILD_WORKBENCH)
 if(NOT UNIX OR APPLE)
  message(FATAL_ERROR "This workbench backend requires Linux X11; other hosts are not qualified")
 endif()
 find_package(Threads REQUIRED)
 find_path(CP_X11_INCLUDE X11/Xlib.h REQUIRED)
 find_library(CP_X11_LIBRARY NAMES X11 libX11.so.6 REQUIRED)
 add_executable(caelum-workbench src/workbench_x11.c)
 target_include_directories(caelum-workbench PRIVATE ${CP_X11_INCLUDE})
 target_link_libraries(caelum-workbench PRIVATE caelum_port ${CP_X11_LIBRARY} Threads::Threads m)
 target_compile_options(caelum-workbench PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math)
 install(TARGETS caelum-workbench RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
